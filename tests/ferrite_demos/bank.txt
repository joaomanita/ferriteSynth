use ferrite_session::{prelude::*};

define_choice! {
    BankOps;
    deposit: ReceiveValue<Int, Z>,
    withdraw: SendValue<Int, ReceiveValue<Int, Z>>,
    Terminate: End
}

type RecBankOps = Rec<ExternalChoice<BankOps>>;

fn bankops(arg1: Int) -> Session<RecBankOps> { @SYNTHESIZE [] }

fn bank_session(withdraw_value: Int) -> Session<ReceiveValue<String, RecBankOps>> { @SYNTHESIZE [use bankops;] }

fn client_session(myId: String, depositAmt: Int) -> Session< ReceiveChannel< ReceiveValue<String, RecBankOps>, End>> { @SYNTHESIZE [] }
