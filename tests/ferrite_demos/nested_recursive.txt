  either::*,
  prelude::*,
};

type Stream = Rec<
  ExternalChoice<Either<
    Rec<InternalChoice<Either<SendValue<String, Z>, S<Z>>>>,
    ReceiveValue<String, End>>
  >
>;

fn user(arg2: String) -> Session<ReceiveChannel<Stream, End>> { @SYNTHESIZE []}

fn main(arg3: String) { @SYNTHESIZE [use producer, user;] }
