use std::format;

use ferrite_session::{
  either::*,
  prelude::*,
};

type myType<A, B, C> = ReceiveChannel<ExternalChoice<Either<ReceiveChannel<A,B>, ReceiveChannel<A, C>>>, ReceiveChannel<A, ExternalChoice<Either<SendChannel<B, End>, SendChannel<C, End>>>>>;

fn myFunc<A, B, C>() -> Session<myType<A, B, C>> where A: Protocol, B: Protocol, C: Protocol { @SYNTHESIZE [] }

fn user<A, B, C>(arg1: A: 1, arg2: ExternalChoice<Either<ReceiveChannel<A,B>, ReceiveChannel<A, C>>>: 1) -> Session<ReceiveChannel<myType<A, B, C>, SendChannel<C, End>>> { @SYNTHESIZE [] }

fn main(arg1: D: 1, arg2: ExternalChoice<Either<ReceiveChannel<D,E>, ReceiveChannel<D, F>>>: 1) -> Session<SendChannel<F, End>> { @SYNTHESIZE [use myFunc, user;] }
