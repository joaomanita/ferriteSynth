use ferrite_session::{
  either::*,
  prelude::*,
};

fn myFunc<A, B, C>() -> Session<myType> { @SYNTHESIZE [] }

fn user<A, B, C>(arg1: A: 1, arg2: ExternalChoice<Either<ReceiveChannel<A,B>, ReceiveChannel<A, C>>>: 1) -> Session<ReceiveChannel<myType, SendChannel<C, End>>> { @SYNTHESIZE [] }

fn main(arg1: A: 1, arg2: ExternalChoice<Either<ReceiveChannel<A,B>, ReceiveChannel<A, C>>>: 1) -> Session<SendChannel<C, End>> { @SYNTHESIZE [use myFunc, user;] }
